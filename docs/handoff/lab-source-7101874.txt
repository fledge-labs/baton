async def mail_event(task, phase, body):
    client=await mail_agent(task['owner']);worker=await mail_agent('codex-worker')
    sender,recipient=(client,worker) if phase=='submitted' else (worker,client)
    await mail_call('macro_contact_handshake',{'project_key':PROJECT,'requester':sender['name'],'target':recipient['name'],'auto_accept':True,'requester_registration_token':sender['token'],'target_registration_token':recipient['token']})
    result=await mail_call('send_message',{'project_key':PROJECT,'sender_name':sender['name'],'sender_token':sender['token'],'to':[recipient['name']],'subject':f'[{phase}] {task["id"]}','body_md':body,'thread_id':task['id']})
    if result.get('count') != 1 or not result.get('verified_sender'):
        raise RuntimeError('Mail delivery not confirmed')
    # Never log the tool result/body. Persist only the shared thread identifier.
    message_ids=[d.get('payload',{}).get('id') for d in result.get('deliveries',[])]
    event = 'mail.failure_reported' if phase == 'failed' else 'mail.' + phase
    audit(event,task_id=task['id'],agent=task['owner'],mail_message_ids=message_ids)
    return result

@mcp.tool()
async def submit_task(idempotency_key: str, prompt: str, ctx: Context) -> dict:
    """Queue a test-only task for the A2A Codex agent. Reusing a key never executes twice."""
    safe_id(idempotency_key)
    if not prompt.strip() or len(prompt)>16000:raise ValueError('Prompt must contain 1..16000 characters')
    agent=owner(ctx)
    task_id=str(uuid.uuid5(uuid.NAMESPACE_URL,PROJECT+'/'+agent+'/'+idempotency_key))
    with database('tasks.sqlite3') as db:
        prior=db.execute('SELECT * FROM tasks WHERE owner=? AND idem=?',(agent,idempotency_key)).fetchone()
        if prior:
            if prior['prompt']!=prompt:raise ValueError('Idempotency key already used for different input')
            audit('task.replay',task_id=task_id,agent=agent,request_id=ctx.request_context.request.headers.get('x-request-id'))
            return {'task_id':task_id,'status':prior['status'],'replay':True}
        db.execute('INSERT INTO tasks(id,owner,idem,prompt,status) VALUES (?,?,?,?,?)',(task_id,agent,idempotency_key,prompt,'queued'))
    audit('task.submitted',task_id=task_id,agent=agent,request_id=ctx.request_context.request.headers.get('x-request-id'),input_sha256=hashlib.sha256(prompt.encode()).hexdigest())
    return {'task_id':task_id,'status':'queued','replay':False}

async def dispatch(task):
    task_id=task['id']
    try:
        await mail_event(task,'submitted',task['prompt'])
    except Exception:
        raise FailureError(Failure('MAIL_SUBMISSION_FAILED')) from None
    audit('task.dispatched',task_id=task_id,agent=task['owner'])
    payload={'jsonrpc':'2.0','id':task_id,'method':'message/send','params':{'message':{'kind':'message','role':'user','messageId':task_id,'parts':[{'kind':'text','text':json.dumps({'task_id':task_id,'owner':task['owner'],'prompt':task['prompt']})}]},'configuration':{'blocking':True}}}
    try:
        token=Path('/run/secrets/dispatcher.token').read_text().strip()
        async with httpx.AsyncClient(timeout=350) as http:
            response=await http.post(GATEWAY+'/a2a/codex',json=payload,headers={'Authorization':'Bearer '+token,'X-Request-ID':task_id,'A2A-Version':'0.3'})
            response.raise_for_status()
    except Exception:
        raise FailureError(Failure('A2A_TRANSPORT_FAILED')) from None
    try:
        result=response.json()
        if not isinstance(result, dict) or 'error' in result:
            raise ValueError('invalid response')
        result = result['result']
        # Our pinned worker SDK issues UUID task IDs. Do not persist arbitrary
        # peer-controlled strings as diagnostic identifiers.
        a2a_id = str(uuid.UUID(result['id']))
    except Exception:
        raise FailureError(Failure('A2A_PROTOCOL_ERROR')) from None
    with database('tasks.sqlite3') as db:
        db.execute('UPDATE tasks SET a2a_id=? WHERE id=?', (a2a_id, task_id))
    status = result.get('status')
    if not isinstance(status, dict):
        raise FailureError(Failure('A2A_PROTOCOL_ERROR'))
    state = status.get('state')
    if state == 'failed':
        raise FailureError(failure_from_a2a(status))
    if state != 'completed':
        raise FailureError(Failure('A2A_TASK_INCOMPLETE'))
    try:
        texts=[p.get('text','') for artifact in result.get('artifacts',[]) for p in artifact.get('parts',[]) if p.get('kind')=='text']
        if not all(isinstance(text, str) for text in texts):
            raise ValueError('invalid result')
    except Exception:
        raise FailureError(Failure('A2A_PROTOCOL_ERROR')) from None
    output='\n'.join(texts)
    if not output.strip():raise FailureError(Failure('A2A_RESULT_MISSING'))
    with database('tasks.sqlite3') as db:
        db.execute("UPDATE tasks SET status='completed',result=?,a2a_id=? WHERE id=?",(output,a2a_id,task_id))
    audit('task.completed',task_id=task_id,agent=task['owner'],a2a_task_id=a2a_id,output_sha256=hashlib.sha256(output.encode()).hexdigest())
    try:
        await mail_event(task,'completed',output)
    except Exception:
        failure = Failure('MAIL_REPORT_FAILED')
        with database('tasks.sqlite3') as db:db.execute('UPDATE tasks SET error=? WHERE id=?',(failure.display(),task_id))
        audit('mail.report_failed', task_id=task_id, a2a_task_id=a2a_id,
              error_code=failure.code, stage=failure.stage)


async def fail_task(task, failure):
    # Commit the original failure before attempting a best-effort mail report.
    with database('tasks.sqlite3') as db:
        db.execute("UPDATE tasks SET status='failed',error=? WHERE id=?", (failure.display(), task['id']))
        row = db.execute('SELECT a2a_id FROM tasks WHERE id=?', (task['id'],)).fetchone()
    a2a_id = row['a2a_id']
    audit('task.failed', task_id=task['id'], agent=task['owner'], a2a_task_id=a2a_id,
          error_code=failure.code, stage=failure.stage)
    try:
        await mail_event(task, 'failed', failure.report())
    except Exception:
        report_failure = Failure('MAIL_REPORT_FAILED')
        audit('mail.report_failed', task_id=task['id'], a2a_task_id=a2a_id,
              error_code=report_failure.code, stage=report_failure.stage,
              original_error_code=failure.code)

async def worker_loop():
    while True:
        with database('tasks.sqlite3') as db:
            row=db.execute("SELECT * FROM tasks WHERE status='queued' ORDER BY rowid LIMIT 1").fetchone()
            if row:db.execute("UPDATE tasks SET status='working' WHERE id=? AND status='queued'",(row['id'],))

            except Exception as e:
                failure = e.failure if isinstance(e, FailureError) else Failure('BRIDGE_INTERNAL_ERROR')
                await fail_task(task, failure)
        await asyncio.sleep(1)

def identity(authorization):
    if not authorization.startswith('Bearer '):
        return None
    digest = hashlib.sha256(authorization[7:].encode()).hexdigest()
    data = json.loads((CONFIG / 'identities.json').read_text())
    return next((v for v in data.values() if v['hash'] == digest and v['active']), None)

class AuthMiddleware:
    def __init__(self, app, role=None):
        self.app, self.role = app, role

    async def __call__(self, scope, receive, send):
        if scope['type'] != 'http':
            return await self.app(scope, receive, send)
        headers = dict(scope.get('headers', []))
        who = identity(headers.get(b'authorization', b'').decode())
        code = 401 if not who else 403 if self.role and who['role'] != self.role else None
        if code:
            audit('auth.denied', status=code, agent=who['agent'] if who else 'unknown')
            await send({'type':'http.response.start', 'status':code, 'headers':[(b'content-type',b'application/json')]})
            await send({'type':'http.response.body','body':b'{"error":"unauthorized"}'})
            return
        scope.setdefault('state', {})['identity'] = who
        await self.app(scope, receive, send)

def _diagnostic_text(stdout, stderr):
    messages = []
    # --json separates CLI error events from prompts, model text and tool output.
    # Never classify arbitrary content from any other event type.
    for line in stdout[-262144:].splitlines():
        try:
            event = json.loads(line)
        except (ValueError, UnicodeDecodeError):
            continue
        if not isinstance(event, dict):
            continue
        detail = event if event.get('type') == 'error' else (
            event.get('error') if event.get('type') == 'turn.failed' else None)
        if isinstance(detail, dict) and isinstance(detail.get('message'), str):
            messages.append(detail['message'][:16384])
    stderr_text = stderr[-65536:].decode('utf-8', errors='replace')
    if re.search(r'(?m)^user\s*$', stderr_text):
        # A human transcript is not an authenticated CLI diagnostic source.
        # --json should never render one; retain only typed JSON errors.
        return '\n'.join(messages).lower()
    panic_lines = 0
    for line in stderr_text.splitlines():
        lowered = line.lower()
        # A normal bundled-bubblewrap fallback is not a sandbox failure.
        if re.search(r'\bwarn(?:ing)?\b|falling back|fallback', lowered):
            continue
        own_error = re.match(r'\s*(?:error:|fatal:|failed to (?:create|initialize) session:|'
                             r'thread [\'"].{0,160} panicked at)', lowered)
        tracing_error = re.search(r'\berror\b\s+(?:codex[_:]|reqwest[_:]|tokio[_:])', lowered)
        if own_error or tracing_error or panic_lines:
            messages.append(line[:16384])
        if 'panicked at' in lowered and own_error:
            panic_lines = 2
        elif panic_lines:
            panic_lines -= 1
    return '\n'.join(messages).lower()

        return Failure('WORKER_MEMORY_EXHAUSTED')

            and re.search(r'(?:spawn|create|start|fork).{0,60}(?:thread|process|session)|'

def worker_exception_failure(error):
    if isinstance(error, FailureError):
        return error.failure
    if isinstance(error, TimeoutError):
        return Failure('CODEX_TIMEOUT')
    if isinstance(error, OSError):
        if error.errno == errno.EAGAIN:
            return Failure('WORKER_RESOURCE_EXHAUSTED')
        if error.errno in (errno.EMFILE, errno.ENFILE):
            return Failure('WORKER_FD_EXHAUSTED')
        if error.errno == errno.ENOMEM:
            return Failure('WORKER_MEMORY_EXHAUSTED')
        if error.errno in (errno.EPERM, errno.EACCES):
            return Failure('CODEX_SANDBOX_FAILED')
        if error.errno == errno.ENOENT:
            return Failure('CODEX_CONFIG_INVALID')
    return Failure('CODEX_EXECUTION_FAILED')


def failure_from_a2a(status):

    parts = message.get('parts') if isinstance(message, dict) else None
    if not isinstance(parts, list):
        return fallback
    accepted = []
    for part in parts:
        if not isinstance(part, dict) or part.get('kind') != 'data':
            continue
        data = part.get('data')
        if not isinstance(data, dict) or 'lab_failure' not in data:

                or not isinstance(payload['code'], str) or payload['code'] not in WORKER_CODES):
            return fallback
        failure = Failure(payload['code'])
        if payload['stage'] != failure.stage:
            return fallback
        accepted.append(failure)
    return accepted[0] if len(accepted) == 1 else fallback

def check_snapshot(*, path=None, now=None):

def write(path, data, mode=0o600):
    path.parent.mkdir(mode=0o700, parents=True, exist_ok=True)
    previous = path.stat() if path.exists() else None
    fd, temporary = tempfile.mkstemp(prefix='.' + path.name + '.', dir=path.parent)
    tmp = Path(temporary)
    try:
        with os.fdopen(fd, 'w') as f:
            os.fchmod(f.fileno(), mode)
            if previous is not None and os.geteuid() == 0:
                os.fchown(f.fileno(), previous.st_uid, previous.st_gid)
            f.write(data)
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp, path)
    finally:
        tmp.unlink(missing_ok=True)


def provision_client(root, agent):
    """The host needs only stdlib; SDK and Mail secrets stay inside the bridge."""
    repo = Path(__file__).resolve().parents[1]
    proof = secrets.token_hex(16)  # A mount challenge, never an authentication token.
    expected = {'agent': agent, 'proof': proof}
    markers = [root / directory / ('.mail-provision-' + proof)
               for directory in ['config', 'data/bridge']]
    try:
        for marker in markers:
            write(marker, json.dumps(expected), 0o644)
        # Existing bridge containers must see BOTH host directories. A Compose
        # project with a different LAB_RUNTIME cannot attest this fresh challenge.
        command = ['docker', 'compose', '--project-directory', str(repo),
                   '--file', str(repo / 'compose.yaml'), 'exec', '-T', 'bridge',
                   'python', '-m', 'lab.provision_mail', '--agent', agent, '--proof', proof]
        result = subprocess.run(command, env={**os.environ, 'LAB_RUNTIME': str(root)},
                                capture_output=True, text=True, timeout=120)
        if result.returncode != 0 or json.loads(result.stdout) != {'status': 'ready', **expected}:
            raise RuntimeError('mail_provisioning_unconfirmed')
    finally:
        for marker in markers:
            marker.unlink(missing_ok=True)


def publish_identity(root, identities, agent, role, token):
    """Keep the delivery file private until both gateway identity files commit."""
    registry = root / 'config/identities.json'
    config = root / 'config/gateway.json'
    dest = root / ('secrets/bridge/dispatcher.token' if role == 'dispatcher'
                   else f'secrets/clients/{agent}.token')
    pending = dest.with_name('.' + dest.name + '.pending')
    previous = {path: path.read_text() if path.exists() else None for path in [registry, config]}
    try:
        write(pending, token + '\n')
        write(registry, json.dumps(identities, indent=2) + '\n', 0o644)
        write(config, json.dumps(gateway(identities), indent=2) + '\n', 0o644)
        os.chmod(root / 'config', 0o755)
        # Preserve the owner's access when replacing a prior delivery file.
        if dest.exists() and os.geteuid() == 0:
            previous_dest = dest.stat()
            os.chown(pending, previous_dest.st_uid, previous_dest.st_gid)
        os.replace(pending, dest)
    except Exception:
        # No new delivery file was published. Restore any partially updated
        # config without changing unrelated identities or an old delivery file.
        for path, content in previous.items():
            try:
                if content is None:
                    path.unlink(missing_ok=True)
                else:
                    write(path, content, 0o644)
            except OSError:
                pass  # The failure remains nonzero; no new token is delivered.
        raise
    finally:
        pending.unlink(missing_ok=True)
    return dest
